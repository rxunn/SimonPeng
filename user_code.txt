// ============ 攝影機運鏡 Keyframes ============
const keyframes = [
  { t: 0.00, pos: [0, 45, 550],   look: [0, 20, 138] },
  { t: 0.08, pos: [0, 25, 300],   look: [0, 15, 100] },
  { t: 0.16, pos: [0, 14, 160],   look: [0, 14, 0] },
  // Station 1: Personal
  { t: 0.26, pos: [-15, 14, 80],  look: [-55, 16, 80] },
  // Station 2: Apps

  // Station 3: Videos
  { t: 0.66, pos: [-15, 14, -40], look: [-55, 16, -40] },
  // Station 4: Design
  { t: 0.86, pos: [15, 14, -90],  look: [55, 16, -90] },
  // Ending
  { t: 0.95, pos: [0, 18, -120],  look: [0, 15, 0] },
  { t: 1.00, pos: [0, 28, -125],  look: [0, 10, 138] }
];

function sampleKeyframes(t){
  if(t <= keyframes[0].t) return { pos: keyframes[0].pos.slice(), look: keyframes[0].look.slice() };
  if(t >= keyframes[keyframes.length-1].t) return { pos: keyframes[keyframes.length-1].pos.slice(), look: keyframes[keyframes.length-1].look.slice() };
  for(let i = 0; i < keyframes.length - 1; i++){
    const a = keyframes[i], b = keyframes[i+1];
    if(t >= a.t && t <= b.t){
      const u = (t - a.t) / (b.t - a.t);
      const e = u < 0.5 ? 4 * u * u * u : 1 - Math.pow(-2 * u + 2, 3) / 2;
      return {
        pos: [ a.pos[0] + (b.pos[0] - a.pos[0]) * e, a.pos[1] + (b.pos[1] - a.pos[1]) * e, a.pos[2] + (b.pos[2] - a.pos[2]) * e ],
        look: [ a.look[0] + (b.look[0] - a.look[0]) * e, a.look[1] + (b.look[1] - a.look[1]) * e, a.look[2] + (b.look[2] - a.look[2]) * e ]
      };
    }
  }
  return { pos: keyframes[0].pos.slice(), look: keyframes[0].look.slice() };
}

const clock = new THREE.Clock();
const camPos = new THREE.Vector3().copy(camera.position);
const camLook = new THREE.Vector3(0, 20, 138);
const tmpPos = new THREE.Vector3();
const tmpLook = new THREE.Vector3();

// ============ RENDER LOOP ============
function render(){
  const dt = Math.min(clock.getDelta(), 0.05);
  const t = clock.getElapsedTime();

  const total = document.documentElement.scrollHeight - window.innerHeight;
  const progress = total > 0 ? Math.min(1, Math.max(0, scrollY / total)) : 0;

  // 控制全息投影容器的透明度，進入大門後 (進度 > 0.16) 才漸漸顯示
  const cssOpacity = Math.max(0, Math.min(1, (progress - 0.16) / 0.06));
  document.getElementById('css-container').style.opacity = cssOpacity;

  // 1. 攝影機運鏡
  const kf = sampleKeyframes(progress);
  tmpPos.set(kf.pos[0], kf.pos[1], kf.pos[2]);
  tmpLook.set(kf.look[0], kf.look[1], kf.look[2]);

  mx += (tmx - mx) * 0.06; my += (tmy - my) * 0.06;
  tmpPos.x += mx * 4; tmpPos.y -= my * 3;

  camPos.lerp(tmpPos, 0.06); camLook.lerp(tmpLook, 0.06);
  camera.position.copy(camPos); camera.lookAt(camLook);

  // 2. 粒子與底座動畫
  pinkParticles.rotation.y = Math.sin(t * 0.1) * 0.05;
  stations.forEach((s, i) => {
    const phase = t + i * 1.5;
    s.core.material.emissiveIntensity = 1.0 + 0.8 * Math.sin(phase * 2.0);
    s.light.intensity = 1.4 + 0.4 * Math.sin(phase * 2.0);
    s.spotLight.intensity = 200 + 80 * Math.sin(phase * 2.0); // 聚光燈也跟著呼吸閃爍
    // 讓全息卡牌的「旋轉門」持續自轉
    s.carousel.rotation.y += dt * 0.15;
  });

  // 3. 【全息卡牌飛行動畫】
  if (activeCard) {
    // 目標位置：攝影機正前方 28 單位距離
    const forward = new THREE.Vector3(0, 0, -28).applyQuaternion(camera.quaternion);
    const targetPos = camera.position.clone().add(forward);
    
    // 平滑飛向鏡頭，並轉正面對著鏡頭
    activeCard.position.lerp(targetPos, 0.1);
    activeCard.quaternion.slerp(camera.quaternion, 0.1);
  }

  // 其他未選取的卡牌，確保它們乖乖待在旋轉門上
  allHoloCards.forEach(card => {
    if (card !== activeCard) {
      if (card.parent === cssScene) {
         // 如果之前是被抽出來的，現在塞回旋轉門裡
         card.userData.originalParent.attach(card);
      }
      // 平滑歸位到原始的排列位置
      card.position.lerp(card.userData.originalPos, 0.1);
      card.quaternion.slerp(card.userData.originalQuat, 0.1);
    }
  });

  // 4. 渲染畫面
  renderer.render(scene, camera);
  cssRenderer.render(cssScene, camera); // 同步渲染 CSS3D
  
  requestAnimationFrame(render);
}
